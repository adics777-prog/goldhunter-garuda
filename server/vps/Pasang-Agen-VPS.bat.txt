@echo off
rem ============================================================================
rem  Pasang GHG VPS Agent: update otomatis EA GOLD HUNTER GARUDA + preset di VPS ini.
rem  Taruh file ini, GHG-VPS-Agent.ps1 dan config.json dalam SATU folder, lalu
rem  klik kanan file ini > Run as administrator.
rem  Agen berjalan tersembunyi dengan hak administrator (Task Scheduler) supaya bisa
rem  menutup & membuka MT5, dan otomatis jalan lagi setiap kali Windows login.
rem  Aman dijalankan ulang: agen lama dihentikan dan diganti.
rem ============================================================================
cd /d "%~dp0"
title Pasang GHG VPS Agent
net session >nul 2>&1
if errorlevel 1 ( echo. & echo Jalankan sebagai Administrator: klik kanan Pasang-Agen-VPS.bat ^> Run as administrator. & echo. & pause & exit /b 1 )
if not exist "GHG-VPS-Agent.ps1" ( echo GHG-VPS-Agent.ps1 tidak ada di folder ini. & pause & exit /b 1 )
if not exist "config.json" ( echo config.json tidak ada. Unduh dari Admin web ^> Pengaturan ^> EA Live ^> VPS. & pause & exit /b 1 )
powershell -NoProfile -ExecutionPolicy Bypass -Command "Get-ChildItem -Path '%~dp0' | Unblock-File"
echo.
echo === 1/4 Hentikan agen lama (bila ada) ===
powershell -NoProfile -ExecutionPolicy Bypass -Command "$me=$PID; Get-CimInstance Win32_Process -Filter \"Name='powershell.exe'\" | Where-Object { $_.ProcessId -ne $me -and $_.CommandLine -match 'GHG-VPS-Agent\.ps1' -and $_.CommandLine -notmatch 'Get-CimInstance' } | ForEach-Object { Stop-Process -Id $_.ProcessId -Force; 'agen lama dihentikan: PID ' + $_.ProcessId }; $lnk=[Environment]::GetFolderPath('Startup')+'\GHG VPS Agent.lnk'; if (Test-Path $lnk) { Remove-Item $lnk -Force; 'shortcut Startup lama dihapus' }"
echo.
echo === 2/4 Uji koneksi ke web (tanpa menyalin apa pun) ===
powershell -NoProfile -ExecutionPolicy Bypass -File "%~dp0GHG-VPS-Agent.ps1" -Once -DryRun
echo.
echo === 3/4 Jalan otomatis setiap Windows login, hak tertinggi (Task Scheduler "GHG VPS Agent") ===
schtasks /Create /F /TN "GHG VPS Agent" /SC ONLOGON /RL HIGHEST /TR "powershell.exe -NoProfile -WindowStyle Hidden -ExecutionPolicy Bypass -File \"%~dp0GHG-VPS-Agent.ps1\""
echo.
echo === 4/4 Menjalankan agen sekarang ===
schtasks /Run /TN "GHG VPS Agent"
echo.
echo SELESAI. Log ada di agent.log di folder ini. Status VPS tampil di Admin web ^> Pengaturan ^> EA Live.
echo Untuk berhenti: buka Task Scheduler, hapus task "GHG VPS Agent", lalu restart Windows.
pause
