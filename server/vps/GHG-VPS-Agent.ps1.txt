#  Uji tanpa menyalin / me-restart:  powershell -ExecutionPolicy Bypass -File GHG-VPS-Agent.ps1 -DryRun -Once
#  Pasang di VPS baru (dipakai installer 1 file):  GHG-VPS-Agent.ps1 -Setup [-Only <id lisensi>]
#    = tutup MetaTrader akun berlisensi, pasang EA + preset, pulihkan backup profil terakhir sebagai profil
#      "GHG Restore", nyalakan Algo Trading + WebRequest (+ daftar URL dari backup), buka MetaTrader lagi.
#    Kode keluar: 0 selesai, 2 MetaTrader belum login di akun lisensi, 3 file EA belum ada di web, 1 gagal.

#  [Experts] (Algo Trading / WebRequest) hanya untuk MT5 - di MT4 atur sekali lewat Tools > Options.

function Log([string]$m) {
  $line = "[{0}] {1}" -f (Get-Date -Format 'yyyy-MM-dd HH:mm:ss'), $m
  Write-Host $line
  try {
    if ((Test-Path $LogFile) -and (Get-Item $LogFile).Length -gt 2MB) { Move-Item $LogFile "$LogFile.old" -Force }
    Add-Content -Path $LogFile -Value $line -Encoding UTF8
  } catch { }
}

function Get-Plat([string]$platform) {
  if ($platform -eq 'mt4') { return @{ mql = 'MQL4'; exe = 'terminal.exe'; proc = 'terminal'; name = 'MT4'; cls = 'MetaQuotes::MetaTrader::4.00'; ext = '.ex4' } }

# Semua folder data MetaTrader (yang punya MQL5\Experts / MQL4\Experts): %APPDATA%\MetaQuotes\Terminal\<id> + folder portable di config
function Get-DataDirs([string]$platform = 'mt5') {
  $Plat = Get-Plat $platform
  $dirs = @()
  $root = Join-Path $env:APPDATA 'MetaQuotes\Terminal'
  if (($Cfg.scan_appdata -ne $false) -and (Test-Path $root)) {
    Get-ChildItem $root -Directory | ForEach-Object { if (Test-Path (Join-Path $_.FullName ($Plat.mql + '\Experts'))) { $dirs += $_.FullName } }

# ---- config\common.ini (MetaTrader 5 menulisnya UTF-16LE): dibaca & ditulis dengan encoding yang sama
function Read-Ini([string]$path) {
  $bytes = [IO.File]::ReadAllBytes($path)
  $enc = New-Object Text.UTF8Encoding($false)
  if ($bytes.Length -ge 2 -and $bytes[0] -eq 0xFF -and $bytes[1] -eq 0xFE) { $enc = New-Object Text.UnicodeEncoding($false, $true) }
  elseif ($bytes.Length -ge 2 -and $bytes[0] -eq 0xFE -and $bytes[1] -eq 0xFF) { $enc = New-Object Text.UnicodeEncoding($true, $true) }
  elseif ($bytes.Length -ge 3 -and $bytes[0] -eq 0xEF -and $bytes[1] -eq 0xBB -and $bytes[2] -eq 0xBF) { $enc = New-Object Text.UTF8Encoding($true) }
  elseif ($bytes.Length -ge 4 -and $bytes[1] -eq 0 -and $bytes[3] -eq 0) { $enc = New-Object Text.UnicodeEncoding($false, $false) }
  $text = $enc.GetString($bytes).TrimStart([char]0xFEFF)
  $lines = New-Object 'System.Collections.Generic.List[string]'
  foreach ($x in ($text -split "`r?`n")) { $lines.Add($x) }
  while ($lines.Count -gt 0 -and $lines[$lines.Count - 1] -eq '') { $lines.RemoveAt($lines.Count - 1) }
  return [pscustomobject]@{ path = $path; enc = $enc; lines = $lines }
}
function Write-Ini($ini) {
  $text = ($ini.lines -join "`r`n") + "`r`n"
  $body = $ini.enc.GetBytes($text); $pre = $ini.enc.GetPreamble()
  $all = New-Object byte[] ($pre.Length + $body.Length)
  [Array]::Copy($pre, 0, $all, 0, $pre.Length); [Array]::Copy($body, 0, $all, $pre.Length, $body.Length)
  [IO.File]::WriteAllBytes($ini.path, $all)
}
# [start, end) baris isi section (tanpa judul); start -1 = section tidak ada
function Ini-Range($ini, [string]$section) {
  $s = -1; $e = $ini.lines.Count
  for ($i = 0; $i -lt $ini.lines.Count; $i++) {
    $x = $ini.lines[$i].Trim()
    if ($x -match '^\[(.+)\]$') { if ($s -ge 0) { $e = $i; break }; if ($Matches[1].Trim() -ieq $section) { $s = $i + 1 } }
  }
  return @($s, $e)
}
function Ini-Get($ini, [string]$section, [string]$key) {
  $r = Ini-Range $ini $section
  if ($r[0] -lt 0) { return $null }
  for ($i = $r[0]; $i -lt $r[1]; $i++) { $x = $ini.lines[$i]; $k = $x.IndexOf('='); if ($k -gt 0 -and $x.Substring(0, $k).Trim() -ieq $key) { return $x.Substring($k + 1).Trim() } }
  return $null
}
function Ini-Lines($ini, [string]$section) {
  $r = Ini-Range $ini $section; $out = @()
  if ($r[0] -lt 0) { return $out }
  for ($i = $r[0]; $i -lt $r[1]; $i++) { if ($ini.lines[$i] -match '^[A-Za-z][A-Za-z0-9_]*=') { $out += $ini.lines[$i].Trim() } }
  return $out
}
function Ini-Set($ini, [string]$section, [string]$key, [string]$value) {
  $r = Ini-Range $ini $section
  if ($r[0] -lt 0) { $ini.lines.Add('[' + $section + ']'); $ini.lines.Add($key + '=' + $value); return }
  for ($i = $r[0]; $i -lt $r[1]; $i++) { $x = $ini.lines[$i]; $k = $x.IndexOf('='); if ($k -gt 0 -and $x.Substring(0, $k).Trim() -ieq $key) { $ini.lines[$i] = $key + '=' + $value; return } }
  $at = $r[1]; while ($at -gt $r[0] -and $ini.lines[$at - 1].Trim() -eq '') { $at-- }
  $ini.lines.Insert($at, $key + '=' + $value)
}

# MetaTrader milik folder instalasi ini. Jalur exe dibaca lewat WMI, lalu Get-Process sebagai cadangan. Bila MetaTrader
# berjalan dengan hak administrator sedangkan agen tidak, jalurnya tidak terbaca: agen harus dipasang ulang (Task Scheduler,
# hak tertinggi) supaya bisa menutup & membuka MetaTrader.
$script:WarnedElev = $false
function Get-TerminalProcs([string]$installDir, [string]$platform = 'mt5') {
  if (-not $installDir) { return @() }
  $Plat = Get-Plat $platform
  $exe = (Join-Path $installDir $Plat.exe).ToLower()
  $all = @(Get-CimInstance Win32_Process -Filter ("Name='{0}'" -f $Plat.exe))
  $hit = @($all | Where-Object { $_.ExecutablePath -and $_.ExecutablePath.ToLower() -eq $exe })
  if ($hit.Count -eq 0 -and $all.Count -gt 0) {
    $ids = @(Get-Process -Name $Plat.proc -ErrorAction SilentlyContinue | Where-Object { $_.Path -and $_.Path.ToLower() -eq $exe } | ForEach-Object { $_.Id })
    $hit = @($all | Where-Object { $ids -contains $_.ProcessId })
    if ($hit.Count -eq 0 -and -not $script:WarnedElev -and @($all | Where-Object { -not $_.ExecutablePath }).Count -gt 0) {
      Log ('PERINGATAN: {0} berjalan dengan hak administrator, agen tidak bisa melihat / me-restart-nya. Jalankan ulang Pasang-Agen-VPS.bat (klik kanan > Run as administrator).' -f $Plat.name)
      $script:WarnedElev = $true
    }
  }
  return $hit
}

function Get-Json([string]$path, $hdr = $Headers) { return Invoke-RestMethod -Uri ($Web + '/api' + $path) -Headers $hdr -TimeoutSec 60 }
function Get-File([string]$path, [string]$out, $hdr = $Headers) { Invoke-WebRequest -Uri ($Web + '/api' + $path) -Headers $hdr -OutFile $out -UseBasicParsing -TimeoutSec 180 }
function Post-Json([string]$path, $obj, $hdr = $Headers) {
  $body = $obj | ConvertTo-Json -Depth 6
  return Invoke-RestMethod -Uri ($Web + '/api' + $path) -Method Post -Headers $hdr -ContentType 'application/json; charset=utf-8' -Body ([Text.Encoding]::UTF8.GetBytes($body)) -TimeoutSec 120
}
function Err-Text($e) {
  $msg = $e.Exception.Message
  if ($e.Exception.Response) {
    try { $msg = "HTTP {0}: {1}" -f [int]$e.Exception.Response.StatusCode, $msg } catch { }
    if ($e.ErrorDetails -and $e.ErrorDetails.Message) { try { $j = $e.ErrorDetails.Message | ConvertFrom-Json; if ($j.error) { $msg += ' - ' + $j.error } } catch { } }
  }
  return $msg
}
function Safe-Name([string]$n) { return ($n -replace '[\\/:*?"<>|]', '').Trim() }
function Cmd-Args($p) {
  $argLine = ''
  if ($p.CommandLine) { $cl = $p.CommandLine.Trim(); if ($cl.StartsWith('"')) { $i = $cl.IndexOf('"', 1); $argLine = $cl.Substring($i + 1).Trim() } else { $i = $cl.IndexOf(' '); if ($i -gt 0) { $argLine = $cl.Substring($i + 1).Trim() } } }
  return $argLine
}
function Start-Terminal([string]$exe, [string]$argLine) {
  if ($argLine) { Start-Process -FilePath $exe -ArgumentList $argLine -WorkingDirectory (Split-Path $exe) } else { Start-Process -FilePath $exe -WorkingDirectory (Split-Path $exe) }
}
# MetaTrader ditutup normal (WM_CLOSE ke jendela utamanya, seperti klik X) supaya chart, EA & inputnya tersimpan.
# $force: bila tidak menutup dalam 2 menit, proses dihentikan paksa (hanya dipakai -Setup).
function Close-Terminal($p, [string]$platform = 'mt5', [bool]$force = $false) {
  $Plat = Get-Plat $platform
  $proc = Get-Process -Id $p.ProcessId -ErrorAction SilentlyContinue
  if (-not $proc) { return $true }
  if (-not [GhgWin]::Close([uint32]$p.ProcessId, $Plat.cls)) { [void]$proc.CloseMainWindow() }
  if ($proc.WaitForExit(120000)) { return $true }
  if (-not $force) { return $false }
  Log ('  {0} PID {1} tidak menutup dalam 2 menit: dihentikan paksa' -f $Plat.name, $p.ProcessId)
  Stop-Process -Id $p.ProcessId -Force -ErrorAction SilentlyContinue
  return $proc.WaitForExit(30000)
}
# ditutup normal, lalu dibuka lagi dengan argumen yang sama
function Restart-Terminal([string]$installDir, [string]$platform = 'mt5') {
  $Plat = Get-Plat $platform
  foreach ($p in (Get-TerminalProcs $installDir $platform)) {
    $exe = $p.ExecutablePath
    $argLine = Cmd-Args $p
    if ($DryRun) { Log ("  [uji] akan me-restart {0} PID {1}: {2} {3}" -f $Plat.name, $p.ProcessId, $exe, $argLine); continue }
    Log ("  menutup {0} PID {1} ({2})" -f $Plat.name, $p.ProcessId, $installDir)
    if (-not (Close-Terminal $p $platform $false)) { Log ('  {0} tidak menutup dalam 2 menit: dibiarkan (restart manual {0} ini agar EA versi baru aktif)' -f $Plat.name); continue }
    Start-Sleep -Seconds 3
    Start-Terminal $exe $argLine
    Log ('  {0} dibuka lagi' -f $Plat.name)
    Start-Sleep -Seconds $Gap
  }
}

# ================================================================ MODE MASTER (admin): semua MT5
function Send-Status([string]$version, [string]$sha, [string]$note, [bool]$installed) {
  $terms = @()
  foreach ($d in (Get-DataDirs)) { $org = Get-Origin $d; $terms += @{ data = $d; origin = $org; running = (@(Get-TerminalProcs $org).Count -gt 0) } }
  try { Post-Json '/vps/status' @{ host = $env:COMPUTERNAME; version = $version; sha = $sha; note = $note; installed = $installed; terminals = $terms } | Out-Null }
  catch { Log ("Status gagal dikirim: " + (Err-Text $_)) }
}

function Install-Release($rel) {
  $tmp = Join-Path $env:TEMP ('ghg-rel-' + ($rel.sha -replace '[^A-Za-z0-9]', '').Substring(0, [Math]::Min(12, ($rel.sha -replace '[^A-Za-z0-9]', '').Length)))
  if (Test-Path $tmp) { Remove-Item $tmp -Recurse -Force }
  New-Item -ItemType Directory -Path $tmp | Out-Null
  $ex5 = Join-Path $tmp $rel.ex5.name
  Get-File ('/vps/file/' + $rel.ex5.id) $ex5
  if ((Get-Item $ex5).Length -ne [int64]$rel.ex5.size) { throw "Ukuran $($rel.ex5.name) tidak cocok (unduhan rusak)" }
  $eas = @($ex5)
  foreach ($x in @($rel.extras)) { if ($x -and $x.name) { $f = Join-Path $tmp $x.name; Get-File ('/vps/file/' + $x.id) $f; if ((Get-Item $f).Length -ne [int64]$x.size) { throw "Ukuran $($x.name) tidak cocok (unduhan rusak)" }; $eas += $f } }
  $sets = @()
  foreach ($p in $rel.presets) { $f = Join-Path $tmp $p.name; Get-File ('/vps/file/' + $p.id) $f; $sets += $f }
  $dirs = Get-DataDirs
  if (-not $dirs) { throw 'Tidak ada folder data MT5 (jalankan MT5 sekali dulu, atau isi extra_data_dirs di config.json)' }
  $origins = @()
  foreach ($d in $dirs) {
    $targets = @((Join-Path $d 'MQL5\Experts'), (Join-Path $d 'MQL5\Presets'), (Join-Path $d 'MQL5\Profiles\Tester'))
    if ($DryRun) { Log ("  [uji] akan menyalin {0} EA + {1} preset ke {2}" -f $eas.Count, $sets.Count, $d) }
    else {
      foreach ($t in $targets) { if (-not (Test-Path $t)) { New-Item -ItemType Directory -Path $t -Force | Out-Null } }
      foreach ($f in $eas) { Copy-Item $f $targets[0] -Force }
      foreach ($f in $sets) { Copy-Item $f $targets[1] -Force; Copy-Item $f $targets[2] -Force }
      Log ("  {0} EA + {1} preset -> {2}" -f $eas.Count, $sets.Count, $d)
    }
    $o = Get-Origin $d
    if ($o -and ($origins -notcontains $o)) { $origins += $o }
  }
  if ($Restart) { foreach ($o in $origins) { Restart-Terminal $o } }
  else { Log '  restart_terminals = false: EA versi baru aktif setelah MT5 di-restart manual' }
  Remove-Item $tmp -Recurse -Force -ErrorAction SilentlyContinue
}

# ================================================================ MODE LISENSI (member): MetaTrader akun berlisensi saja
function Lic-Headers($lic) { return @{ 'x-lic-key' = $lic.key; 'User-Agent' = $UA } }
# folder data platform lisensi + akun yang login di sana; matched = login sama dengan akun lisensi
function Get-LicTerminals($lic) {
  $out = @()
  foreach ($d in (Get-DataDirs $lic.platform)) {
    $org = Get-Origin $d $lic.platform
    $login = Get-DataLogin $d
    $out += [pscustomobject]@{ data = $d; origin = $org; login = $login; matched = ($lic.account -and $login -eq $lic.account) }
  }
  return $out
}
function Send-LicStatus($lic, $st, [string]$note, [bool]$installed) {
  $terms = @()
  foreach ($t in (Get-LicTerminals $lic)) { $terms += @{ data = $t.data; origin = $t.origin; login = $t.login; matched = [bool]$t.matched; running = (@(Get-TerminalProcs $t.origin $lic.platform).Count -gt 0) } }
  try { Post-Json '/vps/lic/status' @{ host = $env:COMPUTERNAME; agent = $AgentVersion; version = [string]$st.version; build_id = [int]$st.build_id; note = $note; installed = $installed; terminals = $terms } (Lic-Headers $lic) | Out-Null }
  catch { Log ("[akun {0}] status gagal dikirim: {1}" -f $lic.account, (Err-Text $_)) }
}

# unduh file EA build terbaru + preset ke folder sementara
function Fetch-LicBuild($lic, $rel) {
  $H = Lic-Headers $lic
  $tmp = Join-Path $env:TEMP ('ghg-lic-' + $lic.id + '-' + $rel.build_id)
  if (Test-Path $tmp) { Remove-Item $tmp -Recurse -Force }
  New-Item -ItemType Directory -Path $tmp | Out-Null
  # nama file tetap dari web (GOLD HUNTER GARUDA_<akun>.ex5) supaya chart tetap menunjuk ke file yang sama
  $name = Safe-Name ([string]$rel.ex5.name)
  if (-not $name) { $name = 'GOLD HUNTER GARUDA_' + $lic.account + (Get-Plat $lic.platform).ext }
  $ea = Join-Path $tmp $name
  Get-File ('/vps/lic/file/' + $rel.ex5.id) $ea $H
  if ((Get-Item $ea).Length -ne [int64]$rel.ex5.size) { throw "Ukuran $name tidak cocok (unduhan rusak)" }
  $sets = @()
  foreach ($p in @($rel.presets)) {
    if (-not $p -or -not $p.name) { continue }
    $f = Join-Path $tmp (Safe-Name ([string]$p.name))
    try { Get-File ('/vps/lic/file/' + $p.id) $f $H; $sets += $f } catch { Log ("  preset {0} dilewati: {1}" -f $p.name, (Err-Text $_)) }
  }
  return [pscustomobject]@{ tmp = $tmp; name = $name; ea = $ea; sets = $sets }
}
# salin EA ke <data>\MQLx\Experts, preset ke Presets (+ Profiles\Tester di MT5)
function Copy-LicBuild($lic, $pkg, $t) {
  $Plat = Get-Plat $lic.platform
  $dirs = @((Join-Path $t.data ($Plat.mql + '\Experts')), (Join-Path $t.data ($Plat.mql + '\Presets')))
  if ($lic.platform -ne 'mt4') { $dirs += (Join-Path $t.data 'MQL5\Profiles\Tester') }
  if ($DryRun) { Log ("  [uji] akan menyalin {0} + {1} preset ke {2} (login {3})" -f $pkg.name, $pkg.sets.Count, $t.data, $(if ($t.login) { $t.login } else { '?' })); return }
  foreach ($x in $dirs) { if (-not (Test-Path $x)) { New-Item -ItemType Directory -Path $x -Force | Out-Null } }
  Copy-Item $pkg.ea $dirs[0] -Force
  foreach ($f in $pkg.sets) { for ($i = 1; $i -lt $dirs.Count; $i++) { Copy-Item $f $dirs[$i] -Force } }
  Log ("  {0} + {1} preset -> {2} (login {3})" -f $pkg.name, $pkg.sets.Count, $t.data, $(if ($t.login) { $t.login } else { '?' }))
}

function Install-LicBuild($lic, $rel) {
  $Plat = Get-Plat $lic.platform
  $terms = @(Get-LicTerminals $lic)
  if (-not $terms) { throw ('Tidak ada folder data {0} (jalankan {0} sekali dan login akun {1}, atau isi extra_data_dirs di config.json)' -f $Plat.name, $lic.account) }
  $pkg = Fetch-LicBuild $lic $rel
  $matched = @($terms | Where-Object { $_.matched })
  $targets = $matched
  if ($matched.Count -eq 0) {
    $targets = $terms
    Log ("  PERHATIAN: tidak ada {0} yang login di akun {1} (dibaca dari config\common.ini). File dipasang di SEMUA {2} folder data {0} (file hanya jalan di akun {1}), tapi tidak ada {0} yang di-restart: restart {0} akun {1} secara manual sekali." -f $Plat.name, $lic.account, $terms.Count)
  }
  $origins = @()
  foreach ($t in $targets) {
    Copy-LicBuild $lic $pkg $t
    if ($t.matched -and $t.origin -and ($origins -notcontains $t.origin)) { $origins += $t.origin }
  }
  if ($Restart) { foreach ($o in $origins) { Restart-Terminal $o $lic.platform } }
  elseif ($origins.Count) { Log ('  restart_terminals = false: EA versi baru aktif setelah {0} di-restart manual' -f $Plat.name) }
  Remove-Item $pkg.tmp -Recurse -Force -ErrorAction SilentlyContinue
  return $matched.Count
}

# Profil aktif: MT5 = <data>\MQL5\Profiles\Charts\<[Charts] ProfileLast di config\common.ini>;
#               MT4 = <data>\profiles\<isi profiles\lastprofile.ini>.
function Get-ProfileDir($t, [string]$platform) {
  if ($platform -eq 'mt4') {
    $name = 'default'; $lp = Join-Path $t.data 'profiles\lastprofile.ini'
    if (Test-Path $lp) { $v = ([IO.File]::ReadAllText($lp)).Trim([char]0, [char]0xFEFF, ' ', "`r", "`n"); if ($v) { $name = $v } }
    return [pscustomobject]@{ name = $name; dir = (Join-Path $t.data ('profiles\' + $name)) }
  }
  $name = 'Default'; $ini = Join-Path $t.data 'config\common.ini'
  if (Test-Path $ini) { $v = Ini-Get (Read-Ini $ini) 'Charts' 'ProfileLast'; if ($v) { $name = $v } }
  return [pscustomobject]@{ name = $name; dir = (Join-Path $t.data ('MQL5\Profiles\Charts\' + $name)) }
}
function Get-ExpertsLines($t, [string]$platform) {
  if ($platform -eq 'mt4') { return @() }
  $ini = Join-Path $t.data 'config\common.ini'
  if (-not (Test-Path $ini)) { return @() }
  return @(Ini-Lines (Read-Ini $ini) 'Experts')
}
# sidik cepat (sha dari nama, ukuran, waktu file + baris [Experts]): file baru dibaca hanya bila berubah
function Get-ProfileFingerprint($t, [string]$platform) {
  $p = Get-ProfileDir $t $platform
  if (-not (Test-Path $p.dir)) { return $null }
  $parts = @($p.name)
  foreach ($f in (Get-ChildItem $p.dir -File | Sort-Object Name)) { $parts += ('{0}|{1}|{2}' -f $f.Name, $f.Length, $f.LastWriteTimeUtc.Ticks) }
  $parts += (Get-ExpertsLines $t $platform)
  return (Sha-Hex ([Text.Encoding]::UTF8.GetBytes(($parts -join "`n"))))
}
function Sha-Hex([byte[]]$b) { $h = [Security.Cryptography.SHA256]::Create(); return (($h.ComputeHash($b) | ForEach-Object { $_.ToString('x2') }) -join '') }
# isi backup: hanya file di folder profil aktif (tanpa sub-folder) + baris [Experts]; tidak pernah accounts.dat / password
function Build-ProfileBackup($t, [string]$platform) {
  $p = Get-ProfileDir $t $platform
  if (-not (Test-Path $p.dir)) { return $null }
  $files = @(); $ghg = 0; $ms = New-Object IO.MemoryStream
  $w = { param([byte[]]$b) $ms.Write($b, 0, $b.Length) }
  & $w ([Text.Encoding]::UTF8.GetBytes($p.name + "`n"))
  foreach ($f in (Get-ChildItem $p.dir -File | Sort-Object Name)) {
    if ($f.Name -notmatch '^[A-Za-z0-9 _.()-]{1,80}$' -or $f.Name.StartsWith('.')) { continue }
    if ($f.Extension -ieq '.chr') {
      $fs = [IO.File]::Open($f.FullName, 'Open', 'Read', 'ReadWrite')
      try { $bytes = [GhgChr]::Process($fs, $true, $null, $null); $ghg += [GhgChr]::LastGhg } finally { $fs.Close() }
    } else {
      if ($f.Length -gt 1MB) { continue }
      $bytes = [IO.File]::ReadAllBytes($f.FullName)
    }
    $files += [pscustomobject]@{ name = $f.Name; b64 = [Convert]::ToBase64String($bytes) }
    & $w ([Text.Encoding]::UTF8.GetBytes($f.Name + "`n")); & $w $bytes
  }
  $experts = @(Get-ExpertsLines $t $platform)
  & $w ([Text.Encoding]::UTF8.GetBytes(($experts -join "`n")))
  $server = ''
  $ini = Join-Path $t.data 'config\common.ini'
  if ($platform -ne 'mt4' -and (Test-Path $ini)) { $server = [string](Ini-Get (Read-Ini $ini) 'Common' 'Server') }
  return [pscustomobject]@{ profile_name = $p.name; files = $files; experts = $experts; ea_charts = $ghg; sha = (Sha-Hex $ms.ToArray()); server = $server
    charts = @($files | Where-Object { $_.name -like '*.chr' }).Count; size = ($files | ForEach-Object { $_.b64.Length } | Measure-Object -Sum).Sum }
}
# tiap ProfEvery menit: profil MetaTrader akun lisensi -> POST /vps/lic/profile (hanya bila berubah, hanya bila ada EA GHG)
function Backup-LicProfile($lic, $profMap) {
  $cands = @(Get-LicTerminals $lic | Where-Object { $_.matched })
  if (-not $cands) { return }
  $cands = @($cands | Sort-Object @{ Expression = { @(Get-TerminalProcs $_.origin $lic.platform).Count -gt 0 }; Descending = $true })
  foreach ($t in $cands) {
    $fp = Get-ProfileFingerprint $t $lic.platform
    if (-not $fp) { continue }
    $last = $profMap[$lic.id]
    if ($last -and $last.fp -eq $fp) { if ($last.ok) { return } else { continue } }
    $b = Build-ProfileBackup $t $lic.platform
    if (-not $b) { continue }
    if ($b.ea_charts -lt 1) {
      Log ("[akun {0}] profil '{1}' belum memuat EA GOLD HUNTER GARUDA: tidak di-backup (backup lama di web tetap)" -f $lic.account, $b.profile_name)
      $profMap[$lic.id] = [pscustomobject]@{ fp = $fp; sha = ''; ok = $false }
      continue
    }
    if ($last -and $last.sha -eq $b.sha) { $profMap[$lic.id] = [pscustomobject]@{ fp = $fp; sha = $b.sha; ok = $true }; Save-LicState; return }
    if ($DryRun) { Log ("[akun {0}] [uji] akan backup profil '{1}' ({2} chart, {3} dengan EA, {4:N0} KB)" -f $lic.account, $b.profile_name, $b.charts, $b.ea_charts, ($b.size / 1024)); return }
    try {
      $r = Post-Json '/vps/lic/profile' @{ host = $env:COMPUTERNAME; agent = $AgentVersion; login = $t.login; server = $b.server; profile_name = $b.profile_name
        files = $b.files; experts_ini = ($b.experts -join "`n"); sha = $b.sha } (Lic-Headers $lic)
      Log ("[akun {0}] backup profil '{1}' terkirim ({2} chart, {3} dengan EA GOLD HUNTER GARUDA, {4:N0} KB{5})" -f $lic.account, $b.profile_name, $b.charts, $b.ea_charts, ($b.size / 1024), $(if ($r.same) { ', sama dengan di web' } else { '' }))
      $profMap[$lic.id] = [pscustomobject]@{ fp = $fp; sha = $b.sha; ok = $true }
      Save-LicState
    } catch {
      $msg = Err-Text $_
      Log ("[akun {0}] backup profil gagal: {1}" -f $lic.account, $msg)
      # ditolak server (terlalu besar / akun lain): tidak dicoba ulang sampai profilnya berubah
      if ($msg -match 'HTTP (400|409|413)') { $profMap[$lic.id] = [pscustomobject]@{ fp = $fp; sha = ''; ok = $false } }
    }
    return
  }
}
# pulihkan backup ke profil "GHG Restore" (MetaTrader harus tertutup) dan jadikan profil aktif
function Restore-LicProfile($lic, $t, $prof, [string]$eaFile) {
  $Plat = Get-Plat $lic.platform
  $eaName = [IO.Path]::GetFileNameWithoutExtension($eaFile)
  $eaPath = $null; if ($lic.platform -ne 'mt4') { $eaPath = 'Experts\' + $eaFile }
  if ($lic.platform -eq 'mt4') { $dir = Join-Path $t.data ('profiles\' + $RestoreProfile) } else { $dir = Join-Path $t.data ('MQL5\Profiles\Charts\' + $RestoreProfile) }
  $n = 0; $ghg = 0
  if ($DryRun) { Log ("  [uji] akan memulihkan {0} file profil '{1}' -> {2}" -f @($prof.files).Count, $prof.profile_name, $dir) }
  else {
    if (Test-Path $dir) { Remove-Item $dir -Recurse -Force }
    New-Item -ItemType Directory -Path $dir -Force | Out-Null
  }
  foreach ($f in @($prof.files)) {
    $name = Safe-Name ([string]$f.name)
    if (-not $name -or $name.StartsWith('.')) { continue }
    $bytes = [Convert]::FromBase64String([string]$f.b64)
    if ($name -like '*.chr') {
      $ms = New-Object IO.MemoryStream(, $bytes)
      $bytes = [GhgChr]::Process($ms, $false, $eaName, $eaPath); $ghg += [GhgChr]::LastGhg
    }
    if (-not $DryRun) { [IO.File]::WriteAllBytes((Join-Path $dir $name), $bytes) }
    $n++
  }
  if (-not $DryRun) {
    if ($lic.platform -eq 'mt4') { [IO.File]::WriteAllText((Join-Path $t.data 'profiles\lastprofile.ini'), $RestoreProfile, [Text.Encoding]::Default) }
    else { $ini = Read-Ini (Join-Path $t.data 'config\common.ini'); Ini-Set $ini 'Charts' 'ProfileLast' $RestoreProfile; Write-Ini $ini }
  }
  Log ("  profil '{0}' (backup {1} dari {2}) dipulihkan sebagai '{3}': {4} file, EA {5} di {6} chart" -f $prof.profile_name, (Get-Date ([DateTimeOffset]::FromUnixTimeSeconds([int64]$prof.updated_at).LocalDateTime) -Format 'yyyy-MM-dd HH:mm'), $prof.host, $RestoreProfile, $n, $eaName, $ghg)
  return $ghg
}
# [Experts] common.ini MT5 (MetaTrader tertutup): Algo Trading + WebRequest aktif, daftar URL WebRequest dari backup
function Set-ExpertsIni($t, $prof) {
  $iniPath = Join-Path $t.data 'config\common.ini'
  $url = $null
  if ($prof -and $prof.experts_ini) { foreach ($x in ([string]$prof.experts_ini -split "`r?`n")) { if ($x -match '^WebRequestUrl=(.+)$') { $url = $Matches[1].Trim() } } }
  if ($DryRun) { Log ("  [uji] akan menulis [Experts] Enabled=1, WebRequest=1{0} di {1}" -f $(if ($url) { ', WebRequestUrl dari backup' } else { '' }), $iniPath); return [bool]$url }
  $ini = Read-Ini $iniPath
  Ini-Set $ini 'Experts' 'Enabled' '1'
  Ini-Set $ini 'Experts' 'WebRequest' '1'
  if ($url) { Ini-Set $ini 'Experts' 'WebRequestUrl' $url }
  Write-Ini $ini
  Log ("  [Experts] Algo Trading = ON, WebRequest = ON{0}" -f $(if ($url) { ', daftar URL WebRequest dari backup' } else { ' (daftar URL belum ada di backup)' }))
  return [bool]$url
}
# baris log terminal (logs\yyyymmdd.log, UTF-16) sejak waktu tertentu yang memuat EA
function Wait-EaLoaded($t, [string]$eaName, [datetime]$since, [int]$seconds = 60) {
  $log = Join-Path $t.data ('logs\' + (Get-Date -Format 'yyyyMMdd') + '.log')
  $until = (Get-Date).AddSeconds($seconds)
  while ((Get-Date) -lt $until) {
    Start-Sleep -Seconds 5
    if (-not (Test-Path $log)) { continue }
    try {
      $fs = [IO.File]::Open($log, 'Open', 'Read', 'ReadWrite'); $sr = New-Object IO.StreamReader($fs, [Text.Encoding]::Unicode, $true); $all = $sr.ReadToEnd(); $sr.Close()
      $hits = @($all -split "`r?`n" | Where-Object { $_ -match [regex]::Escape($eaName) -and $_ -match 'loaded successfully' -and $_ -match '\t(\d\d:\d\d:\d\d)\.' -and ([datetime]::ParseExact($Matches[1], 'HH:mm:ss', $null) -ge $since.AddSeconds(-2)) })
      if ($hits.Count) { return $hits }
    } catch { }
  }
  return @()
}
# MetaTrader terpasang di PC ini (hanya untuk pesan bila belum login)
function Find-Installs([string]$platform) {
  $Plat = Get-Plat $platform; $out = @()
  foreach ($root in @($env:ProgramFiles, ${env:ProgramFiles(x86)})) { if ($root -and (Test-Path $root)) { $out += @(Get-ChildItem $root -Directory -ErrorAction SilentlyContinue | Where-Object { Test-Path (Join-Path $_.FullName $Plat.exe) } | ForEach-Object { $_.FullName }) } }
  return $out
}

  $script:LicMap = @{}; $script:ProfMap = @{}
  if (Test-Path $StateFile) {

function Check-Lic($lic, $map, $last) {
  $st = $map[$lic.id]
  if (-not $st) { $st = [pscustomobject]@{ build_id = ''; version = '' }; $map[$lic.id] = $st }
  $rel = Get-Json ('/vps/lic/release?have=' + [uri]::EscapeDataString([string]$st.build_id)) (Lic-Headers $lic)
  # nomor akun lisensi bisa diganti admin (ganti akun disetujui): ikuti akun dari web, bukan dari config.json lama
  if ($rel.account -and [string]$rel.account -ne $lic.account) { Log ("[akun {0}] lisensi kini untuk akun {1} (dari web)" -f $lic.account, $rel.account); $lic.account = [string]$rel.account }
  $due =-not $last[$lic.id] -or ((Get-Date) - $last[$lic.id]).TotalMinutes -ge 5
  if ($rel.none) {
    if ($due) { Send-LicStatus $lic $st 'belum ada file EA di web' $false; $last[$lic.id] = Get-Date }
  } elseif (-not $rel.same) {
    Log ("[akun {0}] build EA baru #{1} v{2} - memasang..." -f $lic.account, $rel.build_id, $rel.version)
    $n = @(Install-LicBuild $lic $rel)[-1]
    if (-not $DryRun) { $st = [pscustomobject]@{ build_id = [string]$rel.build_id; version = [string]$rel.version }; $map[$lic.id] = $st; Save-LicState $map }
    else { $st = [pscustomobject]@{ build_id = [string]$rel.build_id; version = [string]$rel.version } }
    $note = if ($n -gt 0) { 'terpasang ' + (Get-Date -Format 'yyyy-MM-dd HH:mm') } else { 'terpasang ' + (Get-Date -Format 'yyyy-MM-dd HH:mm') + ', MetaTrader akun ini tidak ditemukan: restart manual' }
    Log ("[akun {0}] build #{1} terpasang" -f $lic.account, $rel.build_id)
    Send-LicStatus $lic $st $note (-not $DryRun)
    $last[$lic.id] = Get-Date
  } elseif ($due) {
    $ok = @(Get-LicTerminals $lic | Where-Object { $_.matched }).Count
    Send-LicStatus $lic $st $(if ($ok -gt 0) { 'ok' } else { 'MetaTrader akun ini tidak ditemukan di VPS (login dulu)' }) $false
    $last[$lic.id] = Get-Date
  }
}

  $t = @($matched | Sort-Object @{ Expression = { @(Get-TerminalProcs $_.origin $lic.platform).Count -gt 0 }; Descending = $true }, @{ Expression = { (Get-Item (Join-Path $_.data 'config\common.ini')).LastWriteTimeUtc }; Descending = $true })[0]
  $res.data = $t.data; $res.terminal = $t.origin
  if (-not $t.origin -or -not (Test-Path (Join-Path $t.origin $Plat.exe))) { $res.message = ('Folder instalasi {0} untuk {1} tidak ditemukan (origin.txt)' -f $Plat.name, $t.data); Log $res.message; return $res }
  if (-not (Test-Path (Join-Path $t.data 'config\accounts.dat'))) { Log ('  PERHATIAN: password akun belum tersimpan di {0} ini (centang "Save password" saat login), EA tidak bisa trading tanpa login.' -f $Plat.name) }
  Log ("[akun {0}] SETUP {1}: {2} | data {3}" -f $lic.account, $Plat.name, $t.origin, $t.data)
  $prof = $null
  try { $prof = Get-Json '/vps/lic/profile' $H; if ($prof.none) { $prof = $null } } catch { Log ('  backup profil tidak bisa diambil: ' + (Err-Text $_)) }
  if ($prof -and $lic.platform -ne [string]$prof.platform -and $prof.platform) { Log ('  backup profil dari {0}, lisensi {1}: tidak dipulihkan' -f $prof.platform, $lic.platform); $prof = $null }
  if ($prof) { Log ("  backup profil terakhir: '{0}' dari {1}, {2} chart ({3} dengan EA)" -f $prof.profile_name, $prof.host, $prof.charts, $prof.ea_charts) } else { Log '  belum ada backup profil di web' }
  $pkg = Fetch-LicBuild $lic $rel
  $res.ea = $pkg.name; $res.version = [string]$rel.version
  # 1) tutup MetaTrader ini (normal, paksa bila menggantung > 2 menit)
  $run = @(Get-TerminalProcs $t.origin $lic.platform)
  $argLine = ''
  foreach ($p in $run) {
    if (-not $argLine) { $argLine = Cmd-Args $p }
    if ($DryRun) { Log ("  [uji] akan menutup {0} PID {1}" -f $Plat.name, $p.ProcessId); continue }
    Log ("  menutup {0} PID {1}..." -f $Plat.name, $p.ProcessId)
    if (-not (Close-Terminal $p $lic.platform $true)) { $res.message = ('{0} tidak bisa ditutup. Tutup {0} secara manual lalu jalankan installer lagi.' -f $Plat.name); Log $res.message; return $res }
  }
  if (-not $argLine -and $t.origin -eq $t.data) { $argLine = '/portable' }
  if (-not $DryRun -and @(Get-TerminalProcs $t.origin $lic.platform).Count) { $res.message = ('{0} masih berjalan. Tutup {0} lalu jalankan installer lagi.' -f $Plat.name); Log $res.message; return $res }
  Start-Sleep -Seconds $(if ($run.Count -and -not $DryRun) { 3 } else { 0 })
  # 2) EA + preset ke semua folder data akun ini
  foreach ($m in $matched) { Copy-LicBuild $lic $pkg $m }
  # 3) profil + [Experts]
  if ($prof -and @($prof.files).Count) {
    $res.ea_charts = Restore-LicProfile $lic $t $prof $pkg.name
    $res.restored = $true; $res.backup_at = [int64]$prof.updated_at; $res.backup_host = [string]$prof.host
  }
  if ($lic.platform -ne 'mt4') { $res.webrequest_urls = Set-ExpertsIni $t $prof }
  # 4) buka lagi
  $since = Get-Date
  $exe = Join-Path $t.origin $Plat.exe
  if ($DryRun) { Log ("  [uji] akan membuka {0} {1}" -f $exe, $argLine) }
  else { Start-Terminal $exe $argLine; Log ("  {0} dibuka: {1} {2}" -f $Plat.name, $exe, $argLine) }
  if (-not $DryRun) {
    $script:LicMap[$lic.id] = [pscustomobject]@{ build_id = [string]$rel.build_id; version = [string]$rel.version }
    $script:ProfMap.Remove($lic.id)
    Save-LicState
    Send-LicStatus $lic $script:LicMap[$lic.id] ('setup ' + (Get-Date -Format 'yyyy-MM-dd HH:mm') + $(if ($res.restored) { ', profil dipulihkan' } else { ', belum ada backup profil' })) $true
    if ($res.restored -and $res.ea_charts -gt 0) {
      $res.loaded = @(Wait-EaLoaded $t ([IO.Path]::GetFileNameWithoutExtension($pkg.name)) $since 75 | ForEach-Object { ($_ -split "`t")[-1] })
      foreach ($x in $res.loaded) { Log ('  ' + $Plat.name + ': ' + $x) }
    }
  }
  Remove-Item $pkg.tmp -Recurse -Force -ErrorAction SilentlyContinue
  $res.code = 0
  if ($res.restored) { $res.message = ('EA {0} v{1} terpasang, setup terakhir ({2} chart, EA di {3} chart) dipulihkan.' -f $pkg.name, $rel.version, $prof.charts, $res.ea_charts) }
  else { $res.message = ('EA {0} v{1} terpasang. Belum ada backup setup di web: pasang EA di chart sekali (drag dari Navigator), agen mem-backup-nya otomatis.' -f $pkg.name, $rel.version) }
  Log ('SELESAI: ' + $res.message)
  return $res
}

    foreach ($t in $terms) { Log ("  [akun {0}] {1}: {2} | data {3} | login {4}{5} | berjalan: {6}" -f $lic.account, $Plat.name, $t.origin, $t.data, $(if ($t.login) { $t.login } else { '?' }), $(if ($t.matched) { ' (COCOK)' } else { '' }), (@(Get-TerminalProcs $t.origin $lic.platform).Count -gt 0)) }
    if ($terms -and -not ($terms | Where-Object { $_.matched })) { Log ("  [akun {0}] PERHATIAN: belum ada {1} yang login di akun ini. Login akun {0} di {1}, lalu agen memasang & me-restart otomatis." -f $lic.account, $Plat.name) }

function Load-State { if (Test-Path $StateFile) { try { $j = Get-Content $StateFile -Raw | ConvertFrom-Json; if ($null -ne $j.sha) { return $j } } catch { } } return [pscustomobject]@{ sha = ''; version = '' } }
function Save-State($st) { $st | ConvertTo-Json | Set-Content -Path $StateFile -Encoding UTF8 }
$state = Load-State
$lastStatus = [datetime]::MinValue
while ($true) {
  try {
    $rel = Get-Json ('/vps/release?have=' + [uri]::EscapeDataString([string]$state.sha))
    if ($rel.none) {
      if (((Get-Date) - $lastStatus).TotalMinutes -ge 5) { Send-Status $state.version $state.sha 'belum ada rilis di web' $false; $lastStatus = Get-Date }
    } elseif (-not $rel.same) {
      Log ("Rilis baru v{0} ({1} EA setup, {2} preset) - memasang..." -f $rel.version, @($rel.extras).Count, @($rel.presets).Count)
      Install-Release $rel
      if (-not $DryRun) { $state = [pscustomobject]@{ sha = $rel.sha; version = $rel.version }; Save-State $state }
      Log ("Rilis v{0} terpasang" -f $rel.version)
      Send-Status $rel.version $rel.sha ('terpasang ' + (Get-Date -Format 'yyyy-MM-dd HH:mm')) (-not $DryRun)
      $lastStatus = Get-Date
    } elseif (((Get-Date) - $lastStatus).TotalMinutes -ge 5) {
      Send-Status $state.version $state.sha 'ok' $false
      $lastStatus = Get-Date
    }
  } catch {
    Log ("Gagal: " + (Err-Text $_))
  }
  if ($Once) { break }
  Start-Sleep -Seconds $Every
}
